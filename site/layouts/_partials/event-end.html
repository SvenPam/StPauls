{{/* An event's end. Missing, unreadable, or before the start falls back to the start. */}} {{- $end := .Date -}} {{- with .Params.end -}} {{- $v := . -}} {{/* Pages CMS writes 2026-10-17T19:30, which time.AsTime only reads with seconds. */}} {{- if findRE `^\d{4}-\d\d-\d\dT\d\d:\d\d$` (string $v) }}{{ $v = printf "%s:00" $v }}{{ end -}} {{- with try (time.AsTime $v) -}} {{- if and (not .Err) (.Value.After $end) }}{{ $end = .Value }}{{ end -}} {{- end -}} {{- end -}} {{- return $end -}}
